   \begin{description}[
     leftmargin=3.0cm,
     labelwidth=2.8cm,
     labelsep=0.2cm,
     style=multiline,
     itemsep=0.6em,
     parsep=0pt,
     topsep=0.2em,
     font=\normalfont\bfseries\sffamily\color{charcoal}%
   ]}
  {\end{description}}
